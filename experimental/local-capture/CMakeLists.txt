cmake_minimum_required(VERSION 3.16)
project(kujiequ_capture_core LANGUAGES CXX)
add_library(kujiequ_capture_core STATIC capture_core.cpp)
target_compile_features(kujiequ_capture_core PUBLIC cxx_std_17)
target_include_directories(kujiequ_capture_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(kujiequ_capture_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(kujiequ_capture_core PUBLIC Threads::Threads)
option(CAPTURE_HOST_TESTS "Build host tests; does not test HarmonyOS" OFF)
if(CAPTURE_HOST_TESTS)
  enable_testing()
  add_executable(capture_core_test capture_core_test.cpp)
  target_link_libraries(capture_core_test PRIVATE kujiequ_capture_core)
  add_test(NAME capture_core_test COMMAND capture_core_test)
endif()
